        function setTransactionType(type) {

            const btnOut = document.getElementById('btn-keluar');
            const nominalInput = document.getElementById('nominal').parentElement;

                document.getElementById('icon-keluar').style.color = 'inherit';

                indicator.style.transform = 'translateX(100%)';

                btnIn.classList.remove('text-primary');

        const menu = document.getElementById('autocomplete-menu');
        const list = document.getElementById('suggestion-list');

        function showSuggestions(val) {
            if (!val) {
                menu.classList.add('hidden');
                return;
            }

            const filtered = suggestions.filter(s => s.toLowerCase().includes(val.toLowerCase()));
            
            if (filtered.length > 0) {
                list.innerHTML = '';
                filtered.forEach(item => {
                    const li = document.createElement('li');
                    li.className = "px-4 py-3 hover:bg-surface-container-low cursor-pointer font-body-md text-body-md text-on-surface";
                    li.textContent = item;
                    li.onclick = () => {
                        document.getElementById('nama-barang').value = item;
                        menu.classList.add('hidden');
                    };
                    list.appendChild(li);
                });
                menu.classList.remove('hidden');
            } else {
                menu.classList.add('hidden');
            }
        }

        // Close menu on click outside

            if (!e.target.closest('#nama-barang')) {

            setTimeout(() => {
                alert('Transaksi berhasil disimpan!');
                btn.innerHTML = originalContent;
                btn.disabled = false;
                e.target.reset();
                setTransactionType(currentType);
            }, 800);

        let startX = 0;

        let activeCard = null;

        function startSwipe(e) {
            startX = e.touches[0].clientX;
            activeCard = e.currentTarget;
            activeCard.style.transition = 'none';
        }

        function moveSwipe(e) {
            if (!activeCard) return;
            const currentX = e.touches[0].clientX;
            const diff = currentX - startX;
            
            // Only allow swipe to left
            if (diff < 0) {
                const translate = Math.max(diff, -80);
                activeCard.style.transform = `translateX(${translate}px)`;
            }
        }

        function endSwipe(e) {
            if (!activeCard) return;
            activeCard.style.transition = 'transform 0.3s cubic-bezier(0.4, 0, 0.2, 1)';
            
            const transform = new WebKitCSSMatrix(window.getComputedStyle(activeCard).transform);
            const currentX = transform.m41;

            if (currentX < -40) {
                activeCard.style.transform = 'translateX(-80px)';
            } else {
                activeCard.style.transform = 'translateX(0px)';
            }
            activeCard = null;
        }

        // Handle click outside to reset swipe
        document.addEventListener('touchstart', (e) => {
            if (activeCard && !activeCard.contains(e.target)) {
                activeCard.style.transform = 'translateX(0px)';
            }
        });
